# ==================================================
# FEATURE EXTRACTION
# ==================================================

def extract_features(y, sr):

    # MFCC
    mfcc = librosa.feature.mfcc(
        y=y,
        sr=sr,
        n_mfcc=20
    )

    mfcc_features = np.mean(
        mfcc,
        axis=1
    )

    # Spectral Centroid
    centroid = librosa.feature.spectral_centroid(
        y=y,
        sr=sr
    )

    centroid_mean = np.mean(centroid)

    # Spectral Bandwidth
    bandwidth = librosa.feature.spectral_bandwidth(
        y=y,
        sr=sr
    )

    bandwidth_mean = np.mean(bandwidth)

    # Zero Crossing Rate
    zcr = librosa.feature.zero_crossing_rate(y)

    zcr_mean = np.mean(zcr)

    # Spectral Rolloff
    rolloff = librosa.feature.spectral_rolloff(
        y=y,
        sr=sr
    )

    rolloff_mean = np.mean(rolloff)

    # Tempo
    tempo, _ = librosa.beat.beat_track(
        y=y,
        sr=sr
    )

    tempo_value = float(
        np.asarray(tempo).flatten()[0]
    )

    # RMS
    rms = librosa.feature.rms(y=y)

    rms_mean = np.mean(rms)

    features = np.concatenate([
        mfcc_features,
        [centroid_mean],
        [bandwidth_mean],
        [zcr_mean],
        [rolloff_mean],
        [tempo_value],
        [rms_mean]
    ])

    return features


# ==================================================
# AUDIO QUALITY ANALYSIS
# ==================================================

def calculate_audio_quality(y):

    # RMS energy
    rms = np.sqrt(
        np.mean(y ** 2)
    )

    # Clipping percentage
    clipping = np.mean(
        np.abs(y) > 0.99
    )

    # Dynamic range
    dynamic_range = np.max(y) - np.min(y)

    # Estimate noise from low-energy samples
    threshold = np.percentile(
        np.abs(y),
        20
    )

    noise_samples = y[
        np.abs(y) <= threshold
    ]

    if len(noise_samples) > 0:

        noise_level = np.std(
            noise_samples
        )

    else:

        noise_level = 0

    # ------------------------------------------
    # Quality score
    # ------------------------------------------

    score = 100

    # Too quiet
    if rms < 0.01:

        score -= 40

    elif rms < 0.03:

        score -= 20

    # Clipping
    if clipping > 0.01:

        score -= 25

    elif clipping > 0.001:

        score -= 10

    # Low dynamic range
    if dynamic_range < 0.2:

        score -= 15

    # Noise
    if noise_level > 0.05:

        score -= 20

    elif noise_level > 0.02:

        score -= 10

    score = max(
        0,
        min(100, score)
    )

    return score, noise_level


# ==================================================
# SIMPLE NOISE REDUCTION
# ==================================================

def reduce_noise(y, sr):

    # High-pass filter
    # Removes very low-frequency unwanted noise

    sos = butter(
        6,
        60,
        btype="highpass",
        fs=sr,
        output="sos"
    )

    cleaned = sosfilt(
        sos,
        y
    )

    # Normalize
    max_value = np.max(
        np.abs(cleaned)
    )

    if max_value > 0:

        cleaned = cleaned / max_value

    return cleaned


# ==================================================
# AUDIO DNA
# ==================================================

def calculate_audio_dna(y, sr):

    tempo, _ = librosa.beat.beat_track(
        y=y,
        sr=sr
    )

    tempo = float(
        np.asarray(tempo).flatten()[0]
    )

    centroid = np.mean(
        librosa.feature.spectral_centroid(
            y=y,
            sr=sr
        )
    )

    bandwidth = np.mean(
        librosa.feature.spectral_bandwidth(
            y=y,
            sr=sr
        )
    )

    rms = np.mean(
        librosa.feature.rms(y=y)
    )

    zcr = np.mean(
        librosa.feature.zero_crossing_rate(y)
    )

    return {
        "Tempo": tempo,
        "Spectral Centroid": centroid,
        "Spectral Bandwidth": bandwidth,
        "Energy": rms,
        "Rhythm / ZCR": zcr
    }


# ==================================================
# EXPLAINABLE AI
# ==================================================

def explain_prediction(model, features):

    importances = model.feature_importances_

    # MFCC importance
    mfcc_importance = np.sum(
        importances[:20]
    )

    # Other feature importance
    acoustic_importance = np.sum(
        importances[20:]
    )

    total = (
        mfcc_importance +
        acoustic_importance
    )

    if total == 0:

        return 50, 50

    mfcc_percentage = (
        mfcc_importance /
        total
    ) * 100

    acoustic_percentage = (
        acoustic_importance /
        total
    ) * 100

    return (
        mfcc_percentage,
        acoustic_percentage
    )


# ==================================================
# GENRE ANALYSIS
# ==================================================

def analyze_genre(features):

    probabilities = model.predict_proba(
        [features]
    )[0]

    genres = model.classes_

    sorted_indices = np.argsort(
        probabilities
    )[::-1]

    results = []

    for index in sorted_indices:

        results.append({
            "genre": genres[index],
            "probability": probabilities[index]
        })

    return results
